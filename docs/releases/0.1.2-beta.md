## Publication commands

```sh
git fetch origin
test "$(git rev-parse HEAD)" = "<approved-commit>"
git diff --quiet -- theme.css manifest.json LICENSE assets/screenshots/archive-olive-512x288.png docs/releases/0.1.2-beta-notes.md
git tag -a 0.1.2 <approved-commit> -m "Archive Olive 0.1.2 beta"
git push origin 0.1.2
gh release create 0.1.2 \
  theme.css \
  manifest.json \
  LICENSE \
  assets/screenshots/archive-olive-512x288.png \
  --repo ivan-94/obsidian-archive-olive \
  --verify-tag \
  --prerelease \
  --title "Archive Olive 0.1.2 Beta" \
  --notes-file docs/releases/0.1.2-beta-notes.md
```
